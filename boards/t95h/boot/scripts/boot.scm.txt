    setenv boot_targets mmc0
    setenv bootargs console=ttyS0,115200 earlycon=uart8250,mmio32,0x05000000 loglevel=8 root=PARTUUID=c5bddd4c-02 rootwait init=/sbin/t95h-diag-init rootfstype=ext4 rw net.ifnames=0
    if load mmc 0:1 0x42000000 /boot/Image; then
        if load mmc 0:1 0x4fa00000 /boot/t95h.dtb; then
            booti 0x42000000 - 0x4fa00000
        fi
    fi
